%lex
%%
\s+ /* skip */
"class" return 'CLASS';
"method" return 'METHOD';
"var" return 'VAR';
"send!" return 'UNSAFESEND';
"send" return 'SEND';
"print" return 'PRINT';
"reply" return 'REPLY';
"new" return 'NEW';
"{" return '{';
"}" return '}';
"\(" return '(';
"\)" return ')';
";" return ';';
"," return ',';
"\." return '.';
"=" return '=';
"\+" return '+';
[0-9]+\b return 'INT';
\"[^\"]*\" return 'STRING';
[a-zA-Z_][a-zA-Z0-9_]* return 'IDENT';
<<EOF>> return 'EOF';
/lex

%start program
%left '+'

%{
import {
  Program, ClassDecl, MethodDecl, Seq,
  VarDecl, Send, Print, Reply,
  NewExpr, Var, IntLit, StringLit, Binop
} from "../ast.js";
%}

%%
program
  : classes stmts EOF	{ return Program($1, $2); }
  ;

classes
  : classes class_decl  { $$ = $1.concat([$2]); }
  | /* nothing */       { $$ = []; }
  ;

class_decl
  : CLASS IDENT '{' methods '}'
    { $$ = ClassDecl($2, $4); }
  ;

methods
  : methods method_decl
    { $$ = $1.concat([$2]); }
  |
    { $$ = []; }
  ;

method_decl
  : METHOD IDENT '(' params ')' '{' stmts '}'
    { $$ = MethodDecl($2, $4, Seq($7)); }
  ;

params
  : IDENT
    { $$ = [$1]; }
  | params ',' IDENT
    { $$ = $1.concat([$3]); }
  |
    { $$ = []; }
  ;

stmts
  : stmts stmt
    { $$ = $1.concat([$2]); }
  |
    { $$ = []; }
  ;

stmt
  : VAR IDENT '=' expr ';'
    { $$ = VarDecl($2, $4); }
  | SEND IDENT '.' IDENT '(' args ')' ';'
    { $$ = Send($2, $4, $6, false); }
  | UNSAFESEND IDENT '.' IDENT '(' args ')' ';'
    { $$ = Send($2, $4, $6, true); }
  | PRINT '(' expr ')' ';'
    { $$ = Print($3); }
  | REPLY '(' expr ')' ';'
    { $$ = Reply($3); }
  ;

args
  : expr
    { $$ = [$1]; }
  | args ',' expr
    { $$ = $1.concat([$3]); }
  |
    { $$ = []; }
  ;

expr
  : INT
    { $$ = IntLit(Number(yytext)); }
  | STRING
    { $$ = StringLit(yytext.slice(1, -1)); }
  | IDENT
    { $$ = Var($1); }
  | NEW IDENT '(' ')'
    { $$ = NewExpr($2); }
  | expr '+' expr
    { $$ = Binop('+', $1, $3); }
  ;
